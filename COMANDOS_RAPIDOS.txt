🚀 INICIO RÁPIDO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Activar entorno:
   $ cd backend
   $ source venv/bin/activate

2. Lanzar dashboard:
   $ streamlit run dashboard.py


📊 EJECUTAR ANÁLISIS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Procesar datos
$ python main.py

# Entrenar modelo
$ python modelo_xgboost.py

# Generar explicabilidad (SHAP)
$ python explicabilidad_xai.py

# Crear recomendaciones
$ python recomendaciones_sistema.py


🔮 GENERAR PREDICCIONES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Por fecha específica
$ python predecir_fecha.py <SEDE> <FECHA> <DIAS>

Ejemplos:
$ python predecir_fecha.py Tunja 2026-02-15 7
$ python predecir_fecha.py Duitama 2026-03-01 10
$ python predecir_fecha.py Sogamoso 2026-04-20 5

# Por sectores
$ python predecir_sectores.py <SEDE> <FECHA> <DIAS>

Ejemplos:
$ python predecir_sectores.py Tunja 2026-02-15 7
$ python predecir_sectores.py Chiquinquirá 2026-03-01 14

🔧 CONFIGURAR RECOMENDACIONES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Editar: backend/config_recomendaciones.json

{
  "umbrales": {
    "pico_consumo": 1.5,        # Ajustar sensibilidad
    "fin_semana": 0.5,
    "nocturno": 0.3
  },
  "costos": {
    "kwh_cop": 1050             # Actualizar tarifa
  },
  "ahorros": {
    "porcentaje_objetivo": 0.15 # Meta de reducción
  }
}

🎯 DEMO RÁPIDA (5 MINUTOS)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. $ streamlit run dashboard.py

2. En dashboard:
   • Mostrar métricas generales
   • Seleccionar "Tunja"
   • Elegir fecha futura (ej: 15 Feb 2026)
   • Configurar tarifa (ej: 1,050 COP/kWh)
   • Click "Generar Predicciones"
   • Expandir "Ver por Sector"
   • Scrollear a "Explicabilidad (XAI)"
   • Ver "Recomendaciones Personalizadas"
